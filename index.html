Your expenses sync across all devices using your own private Supabase database. 100% free, no credit card needed.

  1. Go to supabase.com → New project (sign in with GitHub)
  2. Once created, go to SQL Editor and run the SQL from the grey box below
  3. Go to Project Settings → API
  4. Copy your Project URL and anon / public key
  5. Paste both below and tap Connect
create table expenses (
  id text primary key,
  date text,
  amt numeric,
  cat text,
  note text,
  has_file boolean default false,
  file_name text,
  created_at timestamptz default now()
);

create table settings (
  key text primary key,
  value text
);

alter table expenses enable row level security;
alter table settings enable row level security;
create policy "allow_all" on expenses for all using (true) with check (true);
create policy "allow_all" on settings for all using (true) with check (true);
